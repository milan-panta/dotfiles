cmake_minimum_required(VERSION 3.20)
project(fixture CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_executable(app src/main.cpp src/lib.cpp)
find_package(GTest REQUIRED)
add_executable(lib_test tests/test_lib.cpp src/lib.cpp)
target_link_libraries(lib_test GTest::gtest_main)
target_include_directories(lib_test PRIVATE src)
enable_testing()
include(GoogleTest)
gtest_discover_tests(lib_test)
